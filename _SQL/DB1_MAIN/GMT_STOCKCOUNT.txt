create table if not exists stock_counts (
  id              bigint generated by default as identity primary key,
  branch_key      text,
  branch_name     text,
  counted_by      text,
  user_id         text,
  total_counted   integer default 0,
  total_uncounted integer default 0,
  shortages_count integer default 0,
  surplus_count   integer default 0,
  details         jsonb,          -- تفاصIل الفروقات: [{id,name,barcode,system,actual,diff}]
  created_at      timestamptz default now()
);

create index if not exists idx_stock_counts_branch on stock_counts(branch_key);
create index if not exists idx_stock_counts_date on stock_counts(created_at desc);

-- السماح للـ anon key (نقطة البيع) بالكتابة والقراءة
grant select, insert on stock_counts to anon, authenticated;
grant usage, select on sequence stock_counts_id_seq to anon, authenticated;
