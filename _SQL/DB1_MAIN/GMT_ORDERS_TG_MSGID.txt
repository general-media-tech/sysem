-- GMT_ORDERS_TG_MSGID  ·  2026-07-12
-- الغرض: عمود يحفظ معرّف رسالة تيليغرام الخاصة بكل طلب.
--
-- لماذا؟ طلب المالك: «وقت يتم تعديل أوردر يرجع يبعت رسالة جديدة بالتعديل،
-- وتنمسح الرسالة الأولى إن أمكن».
-- لحذف رسالة من قناة تيليغرام يلزم message_id الخاص بها. بلا هذا العمود لا
-- يستطيع النظام معرفة أي رسالة يحذف، فتبقى على القناة رسالة ببيانات ملغاة
-- (سعر/عنوان/منتجات قديمة) — ومن يجهّز الطلب من القناة يجهّزه غلطاً.
--
-- خُزِّن بالقاعدة لا بذاكرة المتصفح، ليعمل الحذف حتى لو عُدّل الطلب من جهاز آخر.
--
-- آمن للتكرار: يمكن تشغيله أكثر من مرة بلا ضرر.
-- الترتيب: بعد باقي ملفات القاعدة الرئيسية.
-- ══════════════════════════════════════════════════════════════════════

alter table if exists public.gmt_orders
  add column if not exists tg_msg_id bigint;

comment on column public.gmt_orders.tg_msg_id is
  'معرّف رسالة تيليغرام الأخيرة المرسلة عن هذا الطلب — يُستخدم لحذفها عند التعديل. يُدار آلياً.';

-- فهرس خفيف (اختياري) للاستعلام عن الطلبات التي لها رسالة قائمة
create index if not exists idx_gmt_orders_tg_msg_id
  on public.gmt_orders (tg_msg_id)
  where tg_msg_id is not null;

-- ملاحظة تشغيلية مهمة (قيد من تيليغرام لا من النظام):
-- بوتات تيليغرام لا تستطيع حذف رسالة مضى عليها أكثر من 48 ساعة.
-- في هذه الحالة يرسل النظام رسالة التعديل ويكتب فيها صراحةً أنه تعذّر حذف
-- الرسالة السابقة وأن المعتمد هو الرسالة الجديدة — ولا يفشل الحفظ أبداً.
