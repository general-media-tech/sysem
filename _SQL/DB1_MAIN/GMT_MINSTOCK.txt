-- عمود حد التنبيه لكل منتج. القيمة الافتراضية 5 (نفس الحد الثابت القديم)،
-- فالمنتجات الحالية تبقى تعمل بنفس السلوك حتى تخصّص لها حداً مختلفاً.
alter table products add column if not exists min_stock integer default 5;
