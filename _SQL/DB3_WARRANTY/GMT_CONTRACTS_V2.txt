alter table contracts add column if not exists manual_edit boolean default false;   -- هل استُعمل وضع التحرير الحر؟
alter table contracts add column if not exists snapshot_html text;                  -- نسخة النص المطبوع فعلاً
alter table contracts add column if not exists printed_at timestamptz;              -- لحظة الاعتماد/الطباعة
alter table contracts add column if not exists app_version text;                    -- إصدار الأداة

create index if not exists contracts_type_idx on contracts (contract_type);
create index if not exists contracts_created_idx on contracts (created_at desc);
